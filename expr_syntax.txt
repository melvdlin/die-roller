                        | $LiteralExpression
                        | $OperatorExpression

$LiteralExpression      = $DecLiteral

$DecLiteral             = $DecDigit+
$DecDigit               = [0-9]

                        | $DieRollExpression

                        = $Expression - $Expression
                        = $Expression * $Expression
                        = $Expression / $Expression
                        = $Expression % $Expression
