                        | $LiteralExpression
                        | $OperatorExpression

$LiteralExpression      = $DecLiteral

$DecLiteral             = $DecDigit+

$OperatorExpression     = $BinaryExpression
                        | $UnaryExpression

$BinaryExpression       = $Expression $BinaryOperator $ Expression
$UnaryExpression        = $UnaryOperator $Expression

$BinaryOperator         = $ArithmeticOperator | $DieRollOperator
$ArithmeticOperator     = + | - | * | / | %
$DieRollOperator        = d | D

$UnaryOperator          = $NegationOperator
$NegationOperator       = -
